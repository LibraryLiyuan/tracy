cmake_minimum_required(VERSION 3.20)
project(tracy_frame_shortlist_probe LANGUAGES CXX)

# Isolated feasibility executable. It does not change tracy-analyze or rebuild
# the store. The caller supplies the existing, matching Release libraries.
set(TRACY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." CACHE PATH "Tracy source root")
set(TRACY_RELEASE_LIB_DIR "${TRACY_ROOT}/_build-all/query-dev/Release" CACHE PATH "Existing Release libraries")
set(TRACY_JSON_INCLUDE "" CACHE PATH "Directory containing nlohmann/json.hpp")
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This isolated probe currently requires Windows/MSVC")
endif()
if(NOT EXISTS "${TRACY_JSON_INCLUDE}/nlohmann/json.hpp")
    message(FATAL_ERROR "Pass -DTRACY_JSON_INCLUDE pointing to the existing nlohmann headers")
endif()
add_executable(tracy-frame-shortlist-probe main.cpp)
target_compile_features(tracy-frame-shortlist-probe PRIVATE cxx_std_20)
target_compile_definitions(tracy-frame-shortlist-probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(tracy-frame-shortlist-probe PRIVATE "${TRACY_ROOT}/stream/src" "${TRACY_JSON_INCLUDE}")
foreach(lib TracyEventStore TracyStreamCore TracyStoreSqlite TracyStoreCodec TracySnapshotStream)
    if(NOT EXISTS "${TRACY_RELEASE_LIB_DIR}/${lib}.lib")
        message(FATAL_ERROR "Missing prebuilt Release library: ${lib}")
    endif()
    target_link_libraries(tracy-frame-shortlist-probe PRIVATE "${TRACY_RELEASE_LIB_DIR}/${lib}.lib")
endforeach()
target_link_libraries(tracy-frame-shortlist-probe PRIVATE Bcrypt Psapi)
enable_testing()
add_test(NAME frame-shortlist-probe-boundaries COMMAND tracy-frame-shortlist-probe --self-test)
