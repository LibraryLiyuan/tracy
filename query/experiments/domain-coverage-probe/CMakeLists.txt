cmake_minimum_required(VERSION 3.20)
project(tracy_domain_coverage_probe LANGUAGES CXX)
set(TRACY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." CACHE PATH "Tracy root")
set(TRACY_RELEASE_LIB_DIR "${TRACY_ROOT}/_build-all/query-dev/Release" CACHE PATH "Matching Release libraries")
set(TRACY_JSON_INCLUDE "" CACHE PATH "nlohmann include directory")
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${TRACY_JSON_INCLUDE}/nlohmann/json.hpp")
    message(FATAL_ERROR "Use MSVC and specify TRACY_JSON_INCLUDE")
endif()
add_executable(tracy-domain-coverage-probe main.cpp)
target_compile_features(tracy-domain-coverage-probe PRIVATE cxx_std_20)
target_compile_definitions(tracy-domain-coverage-probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(tracy-domain-coverage-probe PRIVATE "${TRACY_ROOT}/stream/src" "${TRACY_JSON_INCLUDE}")
foreach(lib TracyEventStore TracyStreamCore TracyStoreSqlite TracyStoreCodec TracySnapshotStream)
    target_link_libraries(tracy-domain-coverage-probe PRIVATE "${TRACY_RELEASE_LIB_DIR}/${lib}.lib")
endforeach()
target_link_libraries(tracy-domain-coverage-probe PRIVATE Bcrypt)
enable_testing()
add_test(NAME domain-coverage-contract COMMAND tracy-domain-coverage-probe --self-test)
