cmake_minimum_required(VERSION 3.20)
project(tracy-analysis-dictionary-unit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(tracy-analysis-dictionary-tests ../AnalysisDictionaryTests.cpp)
target_include_directories(tracy-analysis-dictionary-tests PRIVATE ../../../analysis)
if(MSVC)
    target_compile_options(tracy-analysis-dictionary-tests PRIVATE /utf-8 /EHsc)
endif()
include(CTest)
add_test(NAME tracy-analysis-dictionary COMMAND tracy-analysis-dictionary-tests)
