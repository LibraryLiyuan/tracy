cmake_minimum_required(VERSION 3.25)
project(tracy-capture-gui LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Capture GUI v1 supports Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS UNICODE _UNICODE)
add_compile_options(/utf-8 /MP "$<$<COMPILE_LANGUAGE:CXX>:/EHsc>")
include(../cmake/CPM.cmake)
CPMAddPackage(NAME json GITHUB_REPOSITORY nlohmann/json GIT_TAG v3.12.0 EXCLUDE_FROM_ALL YES)
CPMAddPackage(NAME glfw GITHUB_REPOSITORY glfw/glfw GIT_TAG 3.4 OPTIONS "GLFW_BUILD_EXAMPLES OFF" "GLFW_BUILD_TESTS OFF" "GLFW_BUILD_DOCS OFF" "GLFW_INSTALL OFF" EXCLUDE_FROM_ALL YES)
CPMAddPackage(NAME ImGui GITHUB_REPOSITORY ocornut/imgui GIT_TAG v1.92.5-docking DOWNLOAD_ONLY YES)
add_library(CaptureGuiImGui STATIC ${ImGui_SOURCE_DIR}/imgui.cpp ${ImGui_SOURCE_DIR}/imgui_draw.cpp ${ImGui_SOURCE_DIR}/imgui_widgets.cpp ${ImGui_SOURCE_DIR}/imgui_tables.cpp ${ImGui_SOURCE_DIR}/misc/cpp/imgui_stdlib.cpp ${ImGui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${ImGui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(CaptureGuiImGui PUBLIC ${ImGui_SOURCE_DIR})
target_link_libraries(CaptureGuiImGui PUBLIC glfw opengl32)
include(../cmake/stream.cmake)
add_library(CaptureGuiCore STATIC src/Core.cpp src/Process.cpp src/Controller.cpp src/Presentation.cpp)
target_include_directories(CaptureGuiCore PUBLIC src)
target_link_libraries(CaptureGuiCore PUBLIC nlohmann_json::nlohmann_json TracyStreamCore bcrypt psapi shell32 ole32 comdlg32 ws2_32)
add_executable(tracy-capture-gui WIN32 src/Main.cpp src/CaptureGui.manifest)
target_link_libraries(tracy-capture-gui PRIVATE CaptureGuiCore CaptureGuiImGui)
include(CTest)
if(BUILD_TESTING)
  add_executable(capture-gui-core-tests tests/CoreTests.cpp)
  target_link_libraries(capture-gui-core-tests PRIVATE CaptureGuiCore)
  add_test(NAME capture-gui-core COMMAND capture-gui-core-tests)
  add_executable(capture-gui-workflow-runner tests/WorkflowRunner.cpp)
  target_link_libraries(capture-gui-workflow-runner PRIVATE CaptureGuiCore)
endif()
install(TARGETS tracy-capture-gui RUNTIME DESTINATION .)
